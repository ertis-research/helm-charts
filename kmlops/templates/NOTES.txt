kMLOps has been deployed in namespace {{ .Release.Namespace }}.

1. Access the kMLOps API (interactive documentation at /docs):
{{- if .Values.httpRoute.enabled }}
  The API is exposed through the HTTPRoute {{ include "kmlops.fullname" . }}.
  {{- range .Values.httpRoute.hostnames }}
  http://{{ . }}/docs
  {{- end }}
{{- else if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "kmlops.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT/docs
{{- else if contains "LoadBalancer" .Values.service.type }}
  It may take a few minutes for the LoadBalancer address to be available. Watch it with:
    kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "kmlops.fullname" . }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "kmlops.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}/docs
{{- else }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "kmlops.fullname" . }} 8000:{{ .Values.service.port }}
  Then open http://localhost:8000/docs
{{- end }}
