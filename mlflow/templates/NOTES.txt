MLflow has been deployed in namespace {{ .Release.Namespace }}.

1. Tracking URI for clients running inside the cluster:
  http://{{ include "mlflow.fullname" . }}.{{ .Release.Namespace }}:{{ .Values.service.port }}

2. Access the MLflow UI:
{{- if .Values.httpRoute.enabled }}
  MLflow is exposed through the HTTPRoute {{ include "mlflow.fullname" . }}.
  {{- range .Values.httpRoute.hostnames }}
  http://{{ . }}
  {{- end }}
{{- else if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "mlflow.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
  Add that address to extraAllowedHosts, or MLflow will answer HTTP 403.
{{- else if contains "LoadBalancer" .Values.service.type }}
  It may take a few minutes for the LoadBalancer address to be available. Watch it with:
    kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "mlflow.fullname" . }}
  Add the external address to extraAllowedHosts, or MLflow will answer HTTP 403.
{{- else }}
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "mlflow.fullname" . }} 5000:{{ .Values.service.port }}
  Then open http://localhost:5000
{{- end }}
